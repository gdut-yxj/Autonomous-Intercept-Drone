// 父类框
//从rclcpp中继承Node类型
class BaseUAV : public rclcpp::Node
{
public:
	//	构造函数 当使用类初始化新的对象时，他会自动执行
	BaseUAV(int uav_id) : Node("offboard_control")
	{
		this->id = uav_id;
		//	创建控制状态发布者 用于发布控制模式的信息
		offboard_control_mode_publisher_ = this->create_publisher<OffboardControlMode>("px4_"+std::to_string(id)+"/fmu/in/offboard_control_mode", 10);
		//	创建轨迹点发布者 用来发布轨迹点信息
		trajectory_setpoint_publisher_ = this->create_publisher<TrajectorySetpoint>("px4_"+std::to_string(id)+"/fmu/in/trajectory_setpoint", 10);
		//	创建命令发布者 用来发布命令信息
		vehicle_command_publisher_ = this->create_publisher<VehicleCommand>("px4_"+std::to_string(id)+"/fmu/in/vehicle_command", 10);

		//	创建速度控制者 用来发布控制速度的信息



		//	初始化轨迹设定点计数器
		offboard_setpoint_counter_ = 0;
	}

	void arm();
	void disarm();

protected:
	virtual void mission_function();
	virtual void publish_offboard_control_mode();
	virtual void publish_trajectory_setpoint(float x,float y,float z,float yaw);
	virtual	void publish_trajectory_velocity(float x,float y,float z,float yaw);
	virtual void publish_vehicle_command(uint16_t command, float param1 = 0.0, float param2 = 0.0);
	int id=0;
	rclcpp::TimerBase::SharedPtr timer_;
	rclcpp::Publisher<OffboardControlMode>::SharedPtr offboard_control_mode_publisher_;
	rclcpp::Publisher<TrajectorySetpoint>::SharedPtr trajectory_setpoint_publisher_;
	rclcpp::Publisher<VehicleCommand>::SharedPtr vehicle_command_publisher_;
	rclcpp::Subscription<px4_msgs::msg::SensorGps>::SharedPtr subscription_;
	std::atomic<uint64_t> timestamp_;   //!< common synced timestamped
	uint64_t offboard_setpoint_counter_;   //!< counter for the number of setpoints sent
private:
};

/**
 * @brief Send a command to Arm the vehicle
 */
 //发送命令对无人机进行解锁
void BaseUAV::arm()
{
	publish_vehicle_command(VehicleCommand::VEHICLE_CMD_COMPONENT_ARM_DISARM, 1.0);

	RCLCPP_INFO(this->get_logger(), "Arm command send");
}

/**
 * @brief Send a command to Disarm the vehicle
 */
void BaseUAV::disarm()
{
	publish_vehicle_command(VehicleCommand::VEHICLE_CMD_COMPONENT_ARM_DISARM, 0.0);

	RCLCPP_INFO(this->get_logger(), "Disarm command send");
}

/**
 * @brief Publish the offboard control mode.
 *        For this example, only position and altitude controls are active.
 */
void BaseUAV::publish_offboard_control_mode()
{
	OffboardControlMode msg{};
	//	启用速度控制
	msg.position = true;
	//	禁用速度控制
	msg.velocity = true;
	//	禁用加速度控制
	msg.acceleration = false;
	//	禁用姿态控制
	msg.attitude = false;
	//	禁用机体角速度控制
	msg.body_rate = false;

	msg.timestamp = this->get_clock()->now().nanoseconds() / 1000;
	offboard_control_mode_publisher_->publish(msg);
}


/**
 * @brief Publish a trajectory setpoint
 *        For this example, it sends a trajectory setpoint to make the
 *        vehicle hover at 5 meters with a yaw angle of 180 degrees.
 */
void BaseUAV::publish_trajectory_setpoint(float x,float y,float z,float yaw)
{
	TrajectorySetpoint msg{};
	//	使用的是FRD坐标系
	msg.position = {x,y,z};
	msg.yaw = yaw; // [-PI:PI]
	msg.timestamp = this->get_clock()->now().nanoseconds() / 1000;
	trajectory_setpoint_publisher_->publish(msg);
}

void BaseUAV::publish_trajectory_velocity(float x,float y,float z,float yaw)
{
	TrajectorySetpoint msg{};
	//	使用的是FRD坐标系
	msg.position = {nan(""),nan(""),nan("")};
	msg.velocity = {x,y,z};
	msg.yaw = yaw; // [-PI:PI]
	msg.timestamp = this->get_clock()->now().nanoseconds() / 1000;
	trajectory_setpoint_publisher_->publish(msg);
}

/**
 * @brief Publish vehicle commands
 * @param command   Command code (matches VehicleCommand and MAVLink MAV_CMD codes)
 * @param param1    Command parameter 1
 * @param param2    Command parameter 2
 */
void BaseUAV::publish_vehicle_command(uint16_t command, float param1, float param2)
{
	VehicleCommand msg{};
	msg.param1 = param1;
	msg.param2 = param2;
	msg.command = command;
	msg.target_system = this->id+1;
	msg.target_component = 1;
	msg.source_system = 1;
	msg.source_component = 1;
	msg.from_external = true;
	msg.timestamp = this->get_clock()->now().nanoseconds() / 1000;
	vehicle_command_publisher_->publish(msg);
}

}


//	逃逸UAV
class UAV_1 : public BaseUAV
{
public:
	UAV_1() : BaseUAV(1){
	this->mission_function();
	rmw_qos_profile_t qos_profile = rmw_qos_profile_sensor_data;
	auto qos = rclcpp::QoS(rclcpp::QoSInitialization(qos_profile.history, 5), qos_profile);
	subscription_ = this->create_subscription<px4_msgs::msg::SensorGps>("px4_"+std::to_string(id)+"/fmu/out/vehicle_gps_position", qos,
	[this](const px4_msgs::msg::SensorGps::SharedPtr msg) {
		UAV1_GPS = msg;
		});
	}
	void mission_function() override;
private:
};
//	逃逸UAV中的重写函数

void UAV_1::mission_function()
{
	//	创建定时器回调函数 每当定时器触发时执行
	auto timer_callback = [this]() -> void {

	if (offboard_setpoint_counter_ == 10) {
		// Change to Offboard mode after 10 setpoints
		this->publish_vehicle_command(VehicleCommand::VEHICLE_CMD_DO_SET_MODE, 1, 6);

		// Arm the vehicle
		this->arm();
	}

	S+=0.1;
	// offboard_control_mode needs to be paired with trajectory_setpoint
	publish_offboard_control_mode();

	// if(S<20)
	// {
	// 	publish_trajectory_setpoint(0, 0, -5.0,-3.14);}
	// else{
	// 	publish_trajectory_setpoint(10*static_cast<float>(sin(S)), 10*static_cast<float>(cos(S)), -5.0,-3.14);}

	publish_trajectory_setpoint(30, 30, -5.0,-3.14);

	// stop the counter after reaching 11
	if (offboard_setpoint_counter_ < 11) {
		offboard_setpoint_counter_++;
	}
	};
	timer_ = this->create_wall_timer(100ms, timer_callback);
}

//	捕捉UAV
class UAV_2 : public BaseUAV
{
public:
	UAV_2() : BaseUAV(2){
	this->mission_function();
	rmw_qos_profile_t qos_profile = rmw_qos_profile_sensor_data;
	auto qos = rclcpp::QoS(rclcpp::QoSInitialization(qos_profile.history, 5), qos_profile);
	subscription_ = this->create_subscription<px4_msgs::msg::SensorGps>("px4_"+std::to_string(id)+"/fmu/out/vehicle_gps_position", qos,
	[this](const px4_msgs::msg::SensorGps::SharedPtr msg) {
		UAV2_GPS = msg;
		});
	}
	void mission_function() override;
private:
};
//	捕捉UAV中的重写函数

void UAV_2::mission_function()
{
	//	创建定时器回调函数 每当定时器触发时执行
	auto timer_callback = [this]() -> void {

	if (offboard_setpoint_counter_ == 10) {
		// Change to Offboard mode after 10 setpoints
		this->publish_vehicle_command(VehicleCommand::VEHICLE_CMD_DO_SET_MODE, 1, 6);
		// Arm the vehicle
		this->arm();
	}
	S+=0.1;
	publish_offboard_control_mode();

	if (S<40)
	{
	publish_trajectory_setpoint(0, 0, -8.0,-3.14);
	}
	else if(S>40)
	{
	double ENU_x,ENU_y,ENU_z;
	//创建 geo_converter
	GeographicLib::LocalCartesian geo_converter(GeographicLib::Constants::WGS84_a(), GeographicLib::Constants::WGS84_f());
	//	设置坐标原点
	geo_converter.Reset(UAV2_GPS->latitude_deg,UAV2_GPS->longitude_deg,UAV2_GPS->altitude_msl_m);
	//	传入GPS坐标
	geo_converter.Forward(UAV1_GPS->latitude_deg,UAV1_GPS->longitude_deg,UAV1_GPS->altitude_msl_m, ENU_x, ENU_y, ENU_z);

	//	NED坐标
	double NED_x,NED_y,NED_z;
	NED_x = ENU_y;
	NED_y = ENU_x;
	NED_z = -ENU_z;
	//	设定速度比率
	double vel = 10;
	//	获取yaw轴
	double yaw = atan2(UAV1_GPS->latitude_deg - UAV2_GPS->latitude_deg,UAV1_GPS->longitude_deg - UAV2_GPS->longitude_deg);

	std::cout<<"####################################"<<std::endl;
	std::cout<<"	UAV1_GPS:"<<UAV1_GPS->latitude_deg<<","<<UAV1_GPS->longitude_deg<<","<<UAV1_GPS->altitude_msl_m<<std::endl;
	std::cout<<"	UAV2_GPS:"<<UAV2_GPS->latitude_deg<<","<<UAV2_GPS->longitude_deg<<","<<UAV2_GPS->altitude_msl_m<<std::endl;
	std::cout<<"		NED1:"<<ENU_x<<","<<ENU_y<<","<<ENU_z<<std::endl;
	std::cout<<"		NED2:"<<0<<","<<0<<","<<0<<std::endl;
	std::cout<<"		ENU1:"<<NED_x<<","<<NED_y<<","<<NED_z<<std::endl;
	std::cout<<"		ENU2:"<<0<<","<<0<<","<<0<<std::endl;


	double m = std::sqrt(NED_x*NED_x+NED_y*NED_y+NED_z*NED_z);
	//	获取模长
	NED_x = NED_x/m*vel;
	NED_y = NED_y/m*vel;
	NED_z = NED_z/m*vel;

	std::cout<<"		after_NED1:"<<ENU_x<<","<<ENU_y<<","<<ENU_z<<std::endl;
	publish_trajectory_velocity(NED_x,NED_y,NED_z,yaw);

	}

	// offboard_control_mode needs to be paired with trajectory_setpoint

	//publish_trajectory_setpoint(30, 30, -5.0,-3.14);
	// stop the counter after reaching 11
	if (offboard_setpoint_counter_ < 11) {
		offboard_setpoint_counter_++;
	}
	};
	timer_ = this->create_wall_timer(100ms, timer_callback);
}


int main(int argc, char *argv[])
{
	std::cout << "Starting offboard control node..." << std::endl;
	setvbuf(stdout, NULL, _IONBF, BUFSIZ);
	rclcpp::init(argc, argv);

//	创建执行器，方便创建多个Node
	rclcpp::executors::MultiThreadedExecutor executor;
//	创建节点
	auto node1 = std::make_shared<UAV_1>();
	auto node2 = std::make_shared<UAV_2>();
//	将节点添加到执行器
	executor.add_node(node1);
    executor.add_node(node2);


	executor.spin();
	rclcpp::shutdown();
	return 0;
}
